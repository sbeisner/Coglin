# Coglin. The zone also prepends Cloudflare's managed content-signals block, so
# the served file has two "User-agent: *" groups — compliant crawlers merge
# groups for the same agent, so both apply. See website/README.md.

User-agent: *
Allow: /

# The application itself is behind a login and has nothing to index.
Disallow: /app/
Disallow: /api/
Disallow: /invite/
Disallow: /reset/
Disallow: /forgot

# /login and /signup are deliberately NOT disallowed, however tempting it looks
# next to the lines above. They are prerendered (src/marketing/seo.ts) and carry
# <meta name="robots" content="noindex">, and a crawler that is not allowed to
# FETCH a page never sees that tag — which is the standard way to end up indexed
# with no snippet instead of not indexed at all. Disallow hides the page from the
# crawler; noindex hides it from the index. These two want the second one.
